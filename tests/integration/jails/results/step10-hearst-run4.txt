######## Appendix: the temporary spec web/e2e/r4-quota.spec.ts as it ran in the repeat (removed after the run, not committed)
import { expect, test } from "@playwright/test";

// TEMPORARY (hearst run 4, step10-progress §27.6 steps 5 and 8): not
// committed. Run through scripts/web-e2e.sh in external mode only.
const env = process.env;
const ready = Boolean(env.R4_USER && env.R4_PASSWORD && env.R4_CID && env.R4_REF && env.R4_TPL && env.R4_TREF);

test("quota below usage and below the template's size, in the edit and create dialogs", async ({ page }) => {
  test.skip(!ready, "hearst run 4 only");
  const errors: string[] = [];
  page.on("pageerror", (err) => errors.push(String(err)));

  // One login for both steps.
  await page.goto(`/containers/${env.R4_CID}`);
  await page.getByLabel("Username").fill(env.R4_USER!);
  await page.getByLabel("Password").fill(env.R4_PASSWORD!);
  await page.getByRole("button", { name: "Log in" }).click();
  await expect(page.getByRole("heading", { name: "r4-q", exact: true })).toBeVisible();

  // Step 5: edit dialog, quota 64M.
  await page.getByRole("button", { name: "Edit", exact: true }).click();
  const edit = page.getByRole("dialog");
  await edit.getByLabel("Disk quota").fill("64M");
  await edit.getByRole("button", { name: "Save" }).click();
  await expect(edit.getByText(new RegExp(`${env.R4_REF} bytes`))).toBeVisible();
  await expect(edit.getByText(/The quota is below what the jail's root holds/)).toBeVisible();
  await expect(edit).toBeVisible();
  (console.log of the message and hint)
  await edit.getByRole("button", { name: "Cancel" }).click();
  await expect(page.getByRole("dialog")).toHaveCount(0);

  // Step 8: create dialog, minimal-151, quota 64M.
  await page.goto("/containers");
  await page.getByRole("button", { name: "Create container" }).click();
  const create = page.getByRole("dialog");
  await create.getByLabel("Name", { exact: true }).fill("r4-q3");   // first run: getByLabel("Name") (matched Hostname too)
  await create.getByLabel("Template").selectOption(env.R4_TPL!);
  await create.getByLabel("Disk quota").fill("64M");
  await create.getByRole("button", { name: "Create" }).click();
  await expect(create.getByText(new RegExp(`${env.R4_TREF} bytes`))).toBeVisible();
  await expect(create.getByText(/for a new jail, the template's size/)).toBeVisible();
  await expect(create).toBeVisible();
  (console.log of the message and hint)
  await create.getByRole("button", { name: "Cancel" }).click();
  await expect(page.getByRole("link", { name: "r4-q3", exact: true })).toHaveCount(0);

  const out = await page.evaluate(async () => (await fetch("/api/v1/auth/logout", { method: "POST" })).status);
  console.log(`logout: ${out}`);
  expect(errors, "page errors").toEqual([]);
});
